cmake_minimum_required(VERSION 3.18)
project(hideout_steam_bridge VERSION 0.1.0 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(HIDEOUT_BRIDGE_SANITIZERS "Enable ASan and UBSan in the isolated harness" OFF)
add_library(bridge_core STATIC src/core.c src/gc_state.c)
set_target_properties(bridge_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(bridge_core PUBLIC include)
target_compile_options(bridge_core PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
if(HIDEOUT_BRIDGE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(bridge_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
add_library(zydis STATIC vendor/zydis/Zydis.c)
target_include_directories(zydis PUBLIC vendor/zydis)
target_compile_definitions(zydis PUBLIC ZYDIS_STATIC_BUILD ZYCORE_STATIC_BUILD ZYDIS_DISABLE_ENCODER ZYDIS_DISABLE_FORMATTER)
add_library(steam_abi STATIC src/steam_abi.cpp)
target_include_directories(steam_abi PUBLIC include)
target_link_libraries(steam_abi PRIVATE zydis)
target_compile_options(steam_abi PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    set(STEAM_ADAPTER src/steam_adapter_i386.c)
  else()
    set(STEAM_ADAPTER src/steam_adapter_x86_64.c)
  endif()
  add_library(hideout_steam_bridge SHARED src/linux_loader.c src/linux_loader_entry.S ${STEAM_ADAPTER} src/linux_transport.c src/linux_host.c src/linux_patch.c src/linux_diagnostics.c)
  target_link_libraries(hideout_steam_bridge PRIVATE bridge_core steam_abi dl pthread rt)
  target_compile_options(hideout_steam_bridge PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
  if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    target_compile_options(hideout_steam_bridge PRIVATE $<$<COMPILE_LANGUAGE:ASM>:-m32>)
  endif()
  target_link_options(hideout_steam_bridge PRIVATE -static-libstdc++ -static-libgcc -Wl,--exclude-libs,ALL,-z,relro,-z,now,-z,noexecstack)
  add_executable(bridge_probe src/bridge_probe.c src/linux_diagnostics.c)
  target_link_libraries(bridge_probe PRIVATE steam_abi dl)
  target_link_options(bridge_probe PRIVATE -static-libstdc++ -static-libgcc -Wl,--exclude-libs,ALL)
  add_executable(bridge_fixture tests/fixture.c src/linux_transport.c)
  target_link_libraries(bridge_fixture PRIVATE bridge_core pthread dl)
  target_compile_options(bridge_fixture PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_executable(bridge_core_test tests/core_test.c)
target_link_libraries(bridge_core_test PRIVATE bridge_core)
target_compile_options(bridge_core_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME bridge_core COMMAND bridge_core_test)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(bridge_session_test tests/session_test.c src/linux_transport.c src/linux_patch.c src/gc_state.c)
  target_include_directories(bridge_session_test PRIVATE include)
  target_link_libraries(bridge_session_test PRIVATE steam_abi pthread)
  target_compile_options(bridge_session_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME bridge_session COMMAND bridge_session_test)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64|i.86")
  foreach(fixture abi_fixture abi_fixture_shifted)
    add_library(${fixture} SHARED tests/abi_fixture.S)
    target_compile_options(${fixture} PRIVATE -m32)
    target_link_options(${fixture} PRIVATE -m32 -nostdlib -Wl,--eh-frame-hdr,--build-id=sha1,-z,notext)
    set_target_properties(${fixture} PROPERTIES LINKER_LANGUAGE C)
  endforeach()
  target_compile_definitions(abi_fixture_shifted PRIVATE ABI_SHIFT)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(bridge_patch_test tests/patch_test.c)
  target_include_directories(bridge_patch_test PRIVATE include)
  target_compile_options(bridge_patch_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME bridge_patch COMMAND bridge_patch_test)
endif()
if(TARGET abi_fixture)
  add_executable(bridge_abi_test tests/abi_test.cpp)
  target_include_directories(bridge_abi_test PRIVATE include)
  target_link_libraries(bridge_abi_test PRIVATE zydis)
  target_compile_options(bridge_abi_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_dependencies(bridge_abi_test abi_fixture abi_fixture_shifted)
  add_test(NAME bridge_abi COMMAND bridge_abi_test $<TARGET_FILE:abi_fixture> $<TARGET_FILE:abi_fixture_shifted>)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT HIDEOUT_BRIDGE_SANITIZERS)
  add_library(loader_dependency SHARED tests/loader_dependency.c)
  set_target_properties(loader_dependency PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/loader-plugins")
  add_library(loader_caller SHARED tests/loader_caller.c)
  target_link_libraries(loader_caller PRIVATE dl)
  set_target_properties(loader_caller PROPERTIES BUILD_RPATH "${CMAKE_CURRENT_BINARY_DIR}/loader-plugins")
  add_executable(bridge_loader_test tests/loader_test.c)
  target_link_libraries(bridge_loader_test PRIVATE loader_caller)
  add_dependencies(bridge_loader_test loader_dependency)
  add_test(NAME bridge_loader COMMAND ${CMAKE_COMMAND} -E env
    "LD_PRELOAD=$<TARGET_FILE:hideout_steam_bridge>" $<TARGET_FILE:bridge_loader_test>)
  add_executable(bridge_preload_test tests/preload_test.c)
  add_test(NAME bridge_preload COMMAND ${CMAKE_COMMAND} -E env
    "LD_PRELOAD=$<TARGET_FILE:hideout_steam_bridge>:libm.so.6"
    "HIDEOUT_STEAM_BRIDGE_LIBRARY=$<TARGET_FILE:hideout_steam_bridge>"
    "HIDEOUT_STEAM_BRIDGE_ENDPOINT=/unused/endpoint" "HIDEOUT_STEAM_BRIDGE_PROBE=/unused/probe"
    $<TARGET_FILE:bridge_preload_test>)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  foreach(fixture elf64_fixture elf64_fixture_high)
    add_library(${fixture} SHARED tests/elf64_fixture.S)

  target_link_options(elf64_fixture_high PRIVATE -Wl,-Ttext-segment=0x100000000)

  target_compile_options(bridge_elf64_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)

endif()
